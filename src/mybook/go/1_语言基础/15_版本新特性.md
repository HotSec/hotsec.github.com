## 一、Go 1.18 (2022-03)

### 1.1 泛型 ⭐⭐⭐

```go
func Min[T constraints.Ordered](a, b T) T {
    if a < b {
        return a
    }
    return b
}

fmt.Println(Min(3, 5))       // int
fmt.Println(Min("a", "b"))   // string

type Stack[T any] struct {
    items []T
}

func (s *Stack[T]) Push(v T) {
    s.items = append(s.items, v)
}

func (s *Stack[T]) Pop() (T, bool) {
    if len(s.items) == 0 {
        var zero T
        return zero, false
    }
    v := s.items[len(s.items)-1]
    s.items = s.items[:len(s.items)-1]
    return v, true
}
```

### 1.2 模糊测试 (Fuzzing)

```go
func FuzzReverse(f *testing.F) {
    testcases := []string{"hello", "世界"}
    for _, tc := range testcases {
        f.Add(tc)
    }
    f.Fuzz(func(t *testing.T, orig string) {
        rev := Reverse(orig)
        doubleRev := Reverse(rev)
        if orig != doubleRev {
            t.Errorf("Reverse twice: got %q, want %q", doubleRev, orig)
        }
    })
}
```

### 1.4 any 和 comparable 别名

```go

    }

func main() {
    // 初始化数据...
    flag.Store(true)
}
```

### 3.2 errors.Join

```go

err2 := errors.New("error 2")

err := errors.Join(err1, err2)
fmt.Println(err)
// error 1
// error 2

// 检查是否包含特定错误

```

### 3.3 PWRAP 错误包装

```go

```

### 3.4 context.WithCancelCause


## 六、Go 1.23 (2024-08)

### 6.1 Iterator 支持 ⭐⭐⭐

```go
// range over func
type Seq[V any] func(yield func(V) bool)

func Backward[E any](s []E) func(yield func(int, E) bool) {
    return func(yield func(int, E) bool) {
        for i := len(s) - 1; i >= 0; i-- {
            if !yield(i, s[i]) {
                return
            }
        }
    }
}

s := []string{"a", "b", "c"}
for i, v := range Backward(s) {
    fmt.Println(i, v)
}
// 2 c
// 1 b
// 0 a
```

### 6.2 unique 包

```go
import "unique"

// 全局字符串去重，节省内存
type Name unique.Handle[string]

func MakeName(s string) Name {
    return Name(unique.Make(s))
}

n1 := MakeName("hello")
n2 := MakeName("hello")
// n1 和 n2 指向同一底层字符串
```

## 七、Go 1.24 (2025-02)

### 7.1 弱指针 (weak pointer)

```go
import "weak"

type Cache struct {
    mu    sync.Mutex
    items map[string]weak.Pointer[[]byte]
}

func (c *Cache) Set(key string, val []byte) {
    c.mu.Lock()
    defer c.mu.Unlock()
    c.items[key] = weak.Make(&val)
}

func (c *Cache) Get(key string) ([]byte, bool) {
    c.mu.Lock()
    defer c.mu.Unlock()
    if wp, ok := c.items[key]; ok {
        if val := wp.Value(); val != nil {
            return *val, true
        }
    }
    return nil, false
}
```

## 十、Go 1.26 (2026-02)

### 10.1 弱指针改进

```go
import "weak"

type Cache[T any] struct {
    mu    sync.Mutex
    items map[string]weak.Pointer[T]
}

func (c *Cache[T]) Set(key string, val T) {
    c.mu.Lock()
    defer c.mu.Unlock()
    c.items[key] = weak.Make(&val)
}

func (c *Cache[T]) Get(key string) (T, bool) {
    c.mu.Lock()
    defer c.mu.Unlock()
    if wp, ok := c.items[key]; ok {
        if val := wp.Value(); val != nil {
            return *val, true
        }
        delete(c.items, key)
    }
    var zero T
    return zero, false
}
```
