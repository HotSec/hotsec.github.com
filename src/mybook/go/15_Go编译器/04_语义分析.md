# 语义分析

> 参考：《Go语言高级编程（第2版）》第5章

语义分析是编译器在语法分析之后进行的处理，负责检查程序的语义正确性，包括作用域分析、变量捕获、逃逸分析等。

func walk(fn *Node) {

        return
    }

## 二、作用域分析

### 2.1 作用域结构

```go
type Scope struct {
    Parent   *Scope
    Children []*Scope
    Vars     map[string]*Node
    Labels   map[string]*Node
}

func newScope(parent *Scope) *Scope {
    return &Scope{
        Parent: parent,
        Vars:   make(map[string]*Node),
        Labels: make(map[string]*Node),
    }
}
```

### 2.2 作用域构建

```go
func buildScopes(fn *Node) {
    scope := newScope(pkgscope)
    fn.Scope = scope
    
    for _, n := range fn.Body.List {
        buildScope(n, scope)
    }
}

func buildScope(n *Node, scope *Scope) {
    switch n.Op {
    case ODCL:
        name := n.Left.Sym.Name
        scope.Vars[name] = n.Left
        
    case OFOR, OFORUNTIL:
        inner := newScope(scope)
        n.Scope = inner
        if n.Init != nil {
            buildScope(n.Init, inner)
        }
        buildScope(n.Body, inner)
        
    case OIF:
        inner := newScope(scope)
        n.Scope = inner
        if n.Init != nil {
            buildScope(n.Init, inner)
        }
        buildScope(n.Body, inner)
        if n.Else != nil {
            buildScope(n.Else, inner)
        }
        
    case OBLOCK:
        inner := newScope(scope)
        n.Scope = inner
        for _, stmt := range n.List {
            buildScope(stmt, inner)
        }
    }
}
```

### 2.3 标识符解析

```go
func resolve(n *Node, scope *Scope) *Node {
    if n.Op != ONAME {
        return n
    }
    
    name := n.Sym.Name
    
    for s := scope; s != nil; s = s.Parent {
        if def, ok := s.Vars[name]; ok {
            n.Name.Def = def
            n.Name.Scope = s
            return n
        }
    }
    
    yyerror("undefined: %s", name)
    return n
}
```

## 三、变量捕获

### 3.1 闭包变量捕获

```go
func capturevars(fn *Node) {
    for _, n := range fn.Body.List {
        if n.Op == OCLOSURE {
            analyzeClosure(n, fn)
        }
    }
}

func analyzeClosure(closure, outer *Node) {
    captured := make(map[*Node]bool)
    
    // 查找闭包中引用的外部变量
    for _, n := range closure.Body.List {
        findCaptures(n, outer.Scope, captured)
    }
    
    // 为每个捕获的变量创建捕获节点
    for v := range captured {
        c := nod(OCAPTURE, v, nil)
        closure.Captures = append(closure.Captures, c)
    }
}

func findCaptures(n *Node, outerScope *Scope, captured map[*Node]bool) {
    if n.Op == ONAME && n.Name.Scope != outerScope {
        captured[n.Name.Def] = true
    }
    
    for _, child := range n.Children() {
        findCaptures(child, outerScope, captured)
    }
}
```

### 3.2 捕获示例

```go
func counter() func() int {
    x := 0
    return func() int {
        x++
        return x
    }
}
```

编译器处理：

```go
func counter() func() int {
    x := new(int)
    *x = 0
    return func() int {
        *x++
        return *x
    }
}
```

### 3.3 值捕获 vs 引用捕获

```go
func example() {
    for i := 0; i < 3; i++ {
        defer func() {
            println(i)
        }()
    }
}
```

输出：
```
3
3
3
```

修复（值捕获）：

```go
func example() {
    for i := 0; i < 3; i++ {
        defer func(i int) {
            println(i)
        }(i)
    }
}
```

输出：
```
2
1
0
```

## 四、逃逸分析

### 4.1 逃逸分析原理

```go
func escape(fn *Node) {
    e := newEscapeAnalysis()
    
    // 1. 分析函数参数
    for _, param := range fn.Type.Params() {
        e.escapes(param)
    }
    
    // 2. 分析函数体
    for _, n := range fn.Body.List {
        e.analyze(n)
    }
    
    // 3. 标记逃逸变量
    for _, v := range e.heap {
        v.Name.Heap = true
    }
}
```

### 4.2 逃逸原因

```go
type escapeAnalysis struct {
    heap    map[*Node]bool
    returns []*Node
}

func (e *escapeAnalysis) analyze(n *Node) {
    switch n.Op {
    case OAS:
        e.analyzeAssign(n)
        
    case ORETURN:
        for _, r := range n.List {
            if e.isAddressOf(r) {
                e.markHeap(r.Left)
            }
        }
        
    case OCALL:
        e.analyzeCall(n)
    }
}

func (e *escapeAnalysis) analyzeAssign(n *Node) {
    // 返回局部变量地址
    if e.isAddressOf(n.Right) && e.isLocal(n.Right.Left) {
        e.markHeap(n.Right.Left)
    }
    
    // 闭包捕获
    if n.Left.Op == OCLOSURE {
        for _, c := range n.Left.Captures {
            if e.isShared(c) {
                e.markHeap(c.Left)
            }
        }
    }
}
```

### 4.3 逃逸分析示例

```go
func escape1() *int {
    x := 1
    return &x
}
```

编译器输出：
```
./main.go:2:2: moved to heap: x
```

```go
func noEscape() int {
    x := 1
    return x
}
```

无逃逸，x 在栈上分配。

### 4.4 查看逃逸分析

```bash

```

输出示例：

## 五、控制流检查

### 5.1 break/continue 检查

```go
func checkBreakContinue(n *Node, loop *Node) {
    switch n.Op {
    case OBREAK:
        if n.Label != nil {
            target := findLabel(n.Label, n.Scope)
            if target == nil {
                yyerror("invalid break label %s", n.Label)
            }
        } else if loop == nil {
            yyerror("break is not in a loop")
        }
        
    case OCONTINUE:
        if n.Label != nil {
            target := findLabel(n.Label, n.Scope)
            if target == nil {
                yyerror("invalid continue label %s", n.Label)
            }
        } else if loop == nil {
            yyerror("continue is not in a loop")
        }
    }
}
```

### 5.2 defer 检查

```go
func checkDefer(n *Node) {
    if n.Op != ODEFER {
        return
    }
    
    // defer 不能在循环中（性能问题）
    if inLoop(n) {
        warn("defer in loop may cause performance issues")
    }
    
    // defer 参数在 defer 语句处求值
    for _, arg := range n.Call.List {
        if isModified(arg) {
            warn("defer argument is modified after defer statement")
        }
    }
}
```

### 5.3 return 路径检查

```go
func checkReturn(fn *Node) {
    if fn.Type.Results().NumFields() == 0 {
        return
    }
    
    if !terminates(fn.Body) {
        yyerror("missing return at end of function")
    }
}

func terminates(n *Node) bool {
    if n == nil {
        return false
    }
    
    switch n.Op {
    case ORETURN, ORECOVER:
        return true
        
    case OBLOCK:
        for _, stmt := range n.List {
            if terminates(stmt) {
                return true
            }
        }
        return false
        
    case OIF:
        return terminates(n.Body) && terminates(n.Else)
        
    case OFOR:
        if n.Cond == nil {
            return false
        }
        return false
        
    default:
        return false
    }
}
```

## 六、初始化顺序

### 6.1 包初始化

```go
func initOrder(pkg *Package) {
    // 1. 收集所有初始化依赖
    deps := make(map[*Node][]*Node)
    for _, n := range pkg.Vars {
        deps[n] = collectDeps(n)
    }
    
    // 2. 拓扑排序
    order := topologicalSort(deps)
    
    // 3. 检查循环依赖
    if hasCycle(deps) {
        yyerror("initialization loop")
    }
    
    // 4. 生成初始化函数
    emitInit(pkg, order)
}
```

### 6.2 初始化顺序示例

```go
var a = b + c
var b = f()
var c = 1

func f() int {
    return c + 1
}
```

初始化顺序：`c → b → a`

## 七、方法集计算

### 7.1 方法集

```go
func methodSet(t *Type) []*Field {
    var methods []*Field
    
    if t.IsStruct() {
        for _, f := range t.Fields() {
            if f.Type.IsMethod() {
                methods = append(methods, f)
            }
        }
    }
    
    if t.IsPointer() {
        for _, m := range methodSet(t.Elem()) {
            if !isPointerReceiver(m) {
                methods = append(methods, m)
            }
        }
    }
    
    return methods
}
```

### 7.2 方法集规则

```
类型 T 的方法集：
- 所有接收者为 T 的方法

类型 *T 的方法集：
- 所有接收者为 T 的方法
- 所有接收者为 *T 的方法
```

## 八、总结

语义分析是编译器的重要阶段，负责：

1. **作用域管理**：变量声明、引用解析
2. **变量捕获**：闭包变量处理
3. **逃逸分析**：栈/堆分配决策
4. **控制流检查**：break/continue/return 验证
5. **初始化顺序**：包级变量初始化
6. **方法集计算**：接口实现检查
