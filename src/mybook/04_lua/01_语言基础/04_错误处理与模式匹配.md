# Lua 错误处理与模式匹配

## 错误处理

### error 与 assert

```lua
local function divide(a, b)
    if b == 0 then
        error("division by zero", 2)
    end
    return a / b
end

local ok, err = pcall(divide, 10, 0)
if not ok then
    print("Error:", err)
end

local function must_be_positive(n)
    assert(n > 0, "value must be positive, got: " .. tostring(n))
    return n
end

local ok, err = pcall(must_be_positive, -5)
if not ok then
    print(err)
end
```

### pcall (Protected Call)

```lua
local function safe_execute(fn, ...)
    local ok, result = pcall(fn, ...)
    if ok then
        return result
    else
        print("Error caught:", result)
        return nil
    end
end

local value = safe_execute(function()
    return tonumber("abc")
end)

local value2 = safe_execute(function()
    error("something went wrong")
end)
```

### xpcall

```lua
local function error_handler(err)
    print("Error handler called:", err)
    print(debug.traceback())
    return err
end

local function risky_operation()
    local data = nil
    return data:method()
end

local ok, result = xpcall(risky_operation, error_handler)
if not ok then
    print("Operation failed:", result)
end
```

### 自定义错误类型

```lua
local Error = {}
Error.__index = Error

function Error.new(code, message)
    return setmetatable({
        code = code,
        message = message,
        timestamp = os.time(),
    }, Error)
end

function Error:__tostring()
    return string.format("[%d] %s", self.code, self.message)
end

function Error:is_retryable()
    return self.code >= 500
end

local ERR_NOT_FOUND = Error.new(404, "resource not found")
local ERR_TIMEOUT = Error.new(504, "request timeout")

local function fetch_resource(id)
    if not id then
        return nil, Error.new(400, "id is required")
    end
    local resource = db_find(id)
    if not resource then
        return nil, ERR_NOT_FOUND
    end
    return resource
end

local resource, err = fetch_resource(nil)
if err then
    print(err)
    if err:is_retryable() then
        print("Will retry...")
    end
end
```

### Result 模式

```lua
local Result = {}
Result.__index = Result

function Result.ok(value)
    return setmetatable({
        is_ok = true,
        value = value,
    }, Result)
end

function Result.err(error)
    return setmetatable({
        is_ok = false,
        error = error,
    }, Result)
end

function Result:unwrap()
    if self.is_ok then
        return self.value
    end
    error("called unwrap on err: " .. tostring(self.error), 2)
end

function Result:unwrap_or(default)
    if self.is_ok then
        return self.value
    end
    return default
end

function Result:map(fn)
    if self.is_ok then
        return Result.ok(fn(self.value))
    end
    return self
end

function Result:and_then(fn)
    if self.is_ok then
        return fn(self.value)
    end
    return self
end

local function parse_int(str)
    local n = tonumber(str)
    if n and math.floor(n) == n then
        return Result.ok(n)
    end
    return Result.err("not a valid integer: " .. str)
end

local result = parse_int("42")
    :map(function(n) return n * 2 end)
    :and_then(function(n)
        if n > 100 then
            return Result.err("too large")
        end
        return Result.ok(n)
    end)

if result.is_ok then
    print("Result:", result.value)
else
    print("Error:", result.error)
end
```

## 模式匹配

### 基础模式

```lua
local s = "Hello, World! 123"

string.find(s, "World")
string.find(s, "%d+")
string.find(s, "^Hello")

local i, j = string.find(s, "World")
print(i, j)

local i, j = string.find(s, "(%d+)")
print(i, j)
```

### 字符类

| 类 | 描述 | 示例 |
|---|------|------|
| `%a` | 字母 | `"a"`, `"Z"` |
| `%d` | 数字 | `"0"`, `"9"` |
| `%l` | 小写字母 | `"a"`, `"z"` |
| `%u` | 大写字母 | `"A"`, `"Z"` |
| `%w` | 字母数字 | `"a"`, `"0"` |
| `%s` | 空白字符 | `" "`, `"\t"` |
| `%p` | 标点 | `","`, `"."` |
| `%c` | 控制字符 | `"\n"`, `"\r"` |
| `.` | 任意字符 | |
| `%x` | 十六进制 | `"a"`, `"F"` |

大写版本表示补集：`%A` 匹配非字母，`%D` 匹配非数字。

### 模式修饰符

| 修饰符 | 描述 |
|--------|------|
| `+` | 1 次或多次（贪婪） |
| `*` | 0 次或多次（贪婪） |
| `-` | 0 次或多次（非贪婪） |
| `?` | 0 次或 1 次 |
| `%b` | 平衡匹配 |
| `^` | 锚定开头 |
| `$` | 锚定结尾 |

### string.match

```lua
local s = "2024-01-15"

local y, m, d = string.match(s, "(%d+)%-(%d+)%-(%d+)")
print(y, m, d)

local date = string.match(s, "%d+%-%d+%-%d+")
print(date)

local email = "user@example.com"
local user, domain = string.match(email, "([^@]+)@([^@]+)")
print(user, domain)
```

### string.gmatch

```lua
local s = "one,two;three four"

for word in string.gmatch(s, "%w+") do
    print(word)
end

local csv = "name,age,city\nAlice,25,Beijing\nBob,30,Shanghai"
for line in string.gmatch(csv, "[^\n]+") do
    local name, age, city = string.match(line, "([^,]+),([^,]+),([^,]+)")
    print(name, age, city)
end
```

### string.gsub

```lua
local s = "hello world"

local result = string.gsub(s, "world", "Lua")
print(result)

local result = string.gsub(s, "(%w+)", function(w)
    return w:sub(1,1):upper() .. w:sub(2)
end)
print(result)

local result = string.gsub("2024-01-15", "(%d+)", function(d)
    return tonumber(d) + 1
end)
print(result)

local result, count = string.gsub("aabbcc", "b", "X")
print(result, count)
```

### 捕获组

```lua
local s = "Hello, World!"

local first, rest = string.match(s, "(%a+),%s*(.*)")
print(first, rest)

local result = string.gsub(s, "(%a+)", "[%1]")
print(result)

local result = string.gsub("abc", "(.)(.)", "%2%1")
print(result)
```

### 平衡匹配 %b

```lua
local s = "a {b {c} d} e"

local inner = string.match(s, "%b{}")
print(inner)

local s2 = "(a (b (c) d) e)"
local inner2 = string.match(s2, "%b()")
print(inner2)

local s3 = "[a [b] c]"
local inner3 = string.match(s3, "%b[]")
print(inner3)
```

### 常用模式示例

```lua
local function is_email(s)
    return string.match(s, "^[%w%.+-]+@[%w.-]+%.[%a]+$") ~= nil
end

local function is_url(s)
    return string.match(s, "^https?://[%w.-]+[/[%w%.%-]*)*$") ~= nil
end

local function is_ipv4(s)
    local p = "(%d+)%.(%d+)%.(%d+)%.(%d+)"
    local a, b, c, d = string.match(s, "^" .. p .. "$")
    if not a then return false end
    local function valid(n) return tonumber(n) <= 255 end
    return valid(a) and valid(b) and valid(c) and valid(d)
end

local function trim(s)
    return string.match(s, "^%s*(.-)%s*$")
end

local function split(str, sep)
    sep = sep or "%s+"
    local result = {}
    for part in string.gmatch(str, "([^" .. sep .. "]+)") do
        result[#result + 1] = part
    end
    return result
end

local function extract_numbers(s)
    local numbers = {}
    for n in string.gmatch(s, "%d+") do
        numbers[#numbers + 1] = tonumber(n)
    end
    return numbers
end

local function camel_to_snake(s)
    return string.gsub(s, "(%u)", function(c)
        return "_" .. c:lower()
    end)
end

local function snake_to_camel(s)
    return string.gsub(s, "_(%a)", function(c)
        return c:upper()
    end)
end
```

### 模式 vs 正则表达式

| 特性 | Lua 模式 | 正则表达式 |
|------|----------|------------|
| 语法 | 简单 | 复杂 |
| 交替 `\|` | 不支持 | 支持 |
| 量词 `{n,m}` | 不支持 | 支持 |
| 非捕获组 | 不支持 | `(?:...)` |
| 前瞻/后顾 | 不支持 | 支持 |
| 性能 | 快 | 较慢 |
| 平衡匹配 | `%b` | 递归模式 |

Lua 模式不是正则表达式，它是更简单但更高效的文本匹配工具。如需完整正则表达式，可使用 `rex_pcre` 或 `lrexlib` 库。
