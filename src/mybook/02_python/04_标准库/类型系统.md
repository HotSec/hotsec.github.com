## 一、基本类型注解

```python
# 基本类型
name: str = "Alice"
age: int = 25
height: float = 1.75
is_student: bool = True

# 容器类型
names: list[str] = ["Alice", "Bob"]
scores: dict[str, int] = {"Alice": 95, "Bob": 87}
coords: tuple[float, float] = (3.14, 2.71)
unique_ids: set[int] = {1, 2, 3}

# 可选类型
from typing import Optional
nickname: Optional[str] = None  # str | None (Python 3.10+)

# 联合类型
from typing import Union
id_value: Union[int, str] = 42  # int | str (Python 3.10+)
```

## 二、函数类型注解

```python
def greet(name: str, times: int = 1) -> str:
    return f"Hello, {name}! " * times

def add(a: int, b: int) -> int:
    return a + b

# 可变参数
def sum_all(*args: int) -> int:
    return sum(args)

# 关键字参数
def configure(**kwargs: str) -> None:
    for k, v in kwargs.items():
        print(f"{k} = {v}")

# 回调函数
from typing import Callable
def process(data: list[int], handler: Callable[[int], str]) -> list[str]:
    return [handler(x) for x in data]
```

## 三、泛型

### 3.1 TypeVar

```python
from typing import TypeVar

T = TypeVar("T")

def first(items: list[T]) -> T:
    return items[0]

# 多 TypeVar
K = TypeVar("K")
V = TypeVar("V")

def get_key(mapping: dict[K, V], key: K) -> V:
    return mapping[key]
```

### 3.2 约束 TypeVar

```python
from typing import TypeVar

Number = TypeVar("Number", int, float)

def add(a: Number, b: Number) -> Number:
    return a + b

add(1, 2)       # ✅
add(1.5, 2.5)   # ✅
# add("a", "b") # ❌ 类型检查失败
```

### 3.3 bound TypeVar

```python
from typing import TypeVar

class Animal:
    def speak(self) -> str: ...

A = TypeVar("A", bound=Animal)

def make_sound(animal: A) -> str:
    return animal.speak()
```

### 3.4 Generic 类

```python
from typing import Generic, TypeVar

T = TypeVar("T")

class Stack(Generic[T]):
    def __init__(self) -> None:
        self._items: list[T] = []

    def push(self, item: T) -> None:
        self._items.append(item)

    def pop(self) -> T:
        return self._items.pop()

    def peek(self) -> T | None:
        return self._items[-1] if self._items else None

s: Stack[int] = Stack()
s.push(42)
value: int = s.pop()
```

## 四、Protocol (结构化类型)

```python
from typing import Protocol

class Drawable(Protocol):
    def draw(self) -> None: ...

class Circle:
    def draw(self) -> None:
        print("Drawing circle")

class Square:
    def draw(self) -> None:
        print("Drawing square")

def render(shape: Drawable) -> None:
    shape.draw()

render(Circle())  # ✅ 鸭子类型，无需显式继承
render(Square())  # ✅
```

## 五、高级类型

### 5.1 Literal

```python
from typing import Literal

def set_mode(mode: Literal["train", "eval", "test"]) -> None:
    ...

set_mode("train")  # ✅
# set_mode("infer") # ❌ 类型检查失败
```

### 5.2 TypedDict

```python
from typing import TypedDict

class UserInfo(TypedDict):
    name: str
    age: int
    email: str

user: UserInfo = {
    "name": "Alice",
    "age": 25,
    "email": "alice@example.com",
}

# 可选字段
class PartialUser(TypedDict, total=False):
    name: str
    age: int
```

### 5.3 Final / LiteralString

```python
from typing import Final

MAX_SIZE: Final[int] = 100
# MAX_SIZE = 200  # ❌ 类型检查警告

class Config:
    VERSION: Final[str] = "1.0"
```

### 5.4 TypeGuard / TypeIs

```python
from typing import TypeGuard

def is_str_list(val: list[object]) -> TypeGuard[list[str]]:
    return all(isinstance(x, str) for x in val)

def process(items: list[object]) -> None:
    if is_str_list(items):
        # items 被收窄为 list[str]
        print(items[0].upper())
```

### 5.5 ParamSpec / Concatenate

```python
from typing import ParamSpec, Callable, TypeVar

P = ParamSpec("P")
R = TypeVar("R")

def retry(fn: Callable[P, R]) -> Callable[P, R]:
    def wrapper(*args: P.args, **kwargs: P.kwargs) -> R:
        for _ in range(3):
            try:
                return fn(*args, **kwargs)
            except Exception:
                continue
        raise RuntimeError("max retries exceeded")
    return wrapper
```

## 六、dataclass 与类型

```python
from dataclasses import dataclass
from typing import ClassVar

@dataclass
class User:
    name: str
    age: int
    email: str = ""
    active: bool = True
    count: ClassVar[int] = 0

    def __post_init__(self) -> None:
        User.count += 1

user = User(name="Alice", age=25)

# frozen dataclass (不可变)
@dataclass(frozen=True)
class Point:
    x: float
    y: float
```
