# Lua GC 与性能优化

## 垃圾回收机制

### GC 算法

Lua 使用**标记-清除 (Mark-and-Sweep)** 增量式垃圾回收器：

1. **标记阶段**：从根集合遍历所有可达对象
2. **清除阶段**：回收不可达对象

Lua 5.1 使用**双白色**机制避免标记阶段的新对象被误回收。Lua 5.2+ 改进为增量式三色标记。

### 三色标记法

| 颜色 | 含义 |
|------|------|
| 白色 | 未访问（候选回收） |
| 灰色 | 已访问但引用未处理 |
| 黑色 | 已访问且引用已处理 |

流程：根对象标灰 → 取灰对象标黑并标引用为灰 → 无灰对象时回收白色对象

### 控制 GC

```lua
collectgarbage("collect")
collectgarbage("count")
collectgarbage("step", 100)
collectgarbage("setpause", 200)
collectgarbage("setstepmul", 200)
collectgarbage("isrunning")

```

### GC 参数

| 参数 | 默认值 | 含义 |
|------|--------|------|
| `pause` | 200 | GC 间隔（内存增长百分比） |
| `stepmul` | 200 | 步进倍率（相对于内存分配速度） |

```lua
collectgarbage("setpause", 150)
collectgarbage("setstepmul", 300)

local mem_before = collectgarbage("count")
for i = 1, 100000 do
    local t = {x = i, y = i * 2}
end
local mem_after = collectgarbage("count")
collectgarbage("collect")
local mem_after_gc = collectgarbage("count")

print(string.format("Before: %.1f KB, After: %.1f KB, After GC: %.1f KB",
    mem_before, mem_after, mem_after_gc))
```

### 分代 GC (Lua 5.4+)

```lua
collectgarbage("generational", 10, 50)

collectgarbage("incremental", 200, 200, 13)
```

分代 GC 假设"新对象更容易被回收"：
- **minor GC**：只回收新生代
- **major GC**：全量回收

### 弱引用类型

```lua
local weak_keys = setmetatable({}, {__mode = "k"})
local weak_values = setmetatable({}, {__mode = "v"})
local weak_both = setmetatable({}, {__mode = "kv"})
```

### 缓存应用

```lua
local Cache = {}
Cache.__index = Cache

function Cache.new()
    return setmetatable({
        _data = setmetatable({}, {__mode = "v"}),
    }, Cache)
end

function Cache:set(key, value)
    self._data[key] = value
end

function Cache:get(key)
    return self._data[key]
end

function Cache:has(key)
    return self._data[key] ~= nil
end

local cache = Cache.new()
cache:set("user:1", {name = "Alice", age = 30})
print(cache:get("user:1"))
collectgarbage("collect")
print(cache:get("user:1") or "collected")
```

### 对象注册表

```lua
local registry = setmetatable({}, {__mode = "k"})

local obj = {id = 1, name = "test"}
registry[obj] = true

obj = nil
collectgarbage("collect")

local count = 0
for _ in pairs(registry) do count = count + 1 end
print("Alive objects:", count)
```

## 终结器 (__gc)

```lua
local File = {}
File.__index = File

function File.new(path, mode)
    local fh, err = io.open(path, mode or "r")
    if not fh then return nil, err end

    local self = setmetatable({
        handle = fh,
        path = path,
    }, File)
    return self
end

function File:read(...)
    return self.handle:read(...)
end

function File:write(...)
    return self.handle:write(...)
end

function File:close()
    if self.handle then
        self.handle:close()
        self.handle = nil
    end
end

function File:__gc()
    self:close()
end

do
    local f = File.new("test.txt", "w")
    f:write("hello\n")
end

collectgarbage("collect")
```

## 性能优化技巧

### 1. 局部变量优先

```lua
local sin = math.sin
local sqrt = math.sqrt

for i = 1, 1000000 do
    local x = sin(i)
    local y = sqrt(i)
end
```

### 2. 避免字符串拼接

```lua
local parts = {}
for i = 1, 10000 do
    parts[#parts + 1] = tostring(i)
end
local result = table.concat(parts, ",")

local result = ""
for i = 1, 10000 do
    result = result .. tostring(i) .. ","
end
```

### 3. 预分配表大小

```lua
local t = {}
for i = 1, 10000 do
    t[i] = i * 2
end

local t = {nil, nil, nil}
```

### 4. 使用 numeric for 替代 generic for

```lua
local t = {1, 2, 3, 4, 5}
local sum = 0
for i = 1, #t do
    sum = sum + t[i]
end

local sum = 0
for _, v in ipairs(t) do
    sum = sum + v
end
```

### 5. 避免频繁创建闭包

```lua
local function make_handler(action)
    return function(data)
        return action(data)
    end
end

local handlers = {
    process = make_handler(process_data),
    validate = make_handler(validate_data),
}

local function process_item(item)
    process_data(item)
end

local function validate_item(item)
    validate_data(item)
end
```

### 6. 字符串池化

```lua
local STR_SUCCESS = "success"
local STR_ERROR = "error"
local STR_PENDING = "pending"

local function get_status(code)
    if code == 0 then return STR_SUCCESS end
    if code < 0 then return STR_ERROR end
    return STR_PENDING
end
```

### 7. 避免全局变量

```lua
local max = math.max
local min = math.min
local format = string.format

local function clamp(v, lo, hi)
    return max(lo, min(v, hi))
end
```

### 8. 使用 memoization

```lua
local memo = setmetatable({}, {__mode = "kv"})

local function fibonacci(n)
    if n <= 1 then return n end
    if memo[n] then return memo[n] end
    local result = fibonacci(n - 1) + fibonacci(n - 2)
    memo[n] = result
    return result
end
```

### 9. 减少 GC 压力

```lua
local obj_pool = {}

local function acquire()
    local obj = table.remove(obj_pool)
    if not obj then
        obj = {x = 0, y = 0, z = 0}
    end
    obj.x, obj.y, obj.z = 0, 0, 0
    return obj
end

local function release(obj)
    obj_pool[#obj_pool + 1] = obj
end

local function process()
    local obj = acquire()
    obj.x = 100
    release(obj)
end
```

## 性能分析

### 简单计时

```lua
local function benchmark(name, fn, iterations)
    iterations = iterations or 100000
    local start = os.clock()
    for i = 1, iterations do
        fn(i)
    end
    local elapsed = os.clock() - start
    print(string.format("%s: %.4fs (%d iterations, %.2f ns/op)",
        name, elapsed, iterations, elapsed / iterations * 1e9))
end

benchmark("table insert", function(i)
    local t = {}
    t[1] = i
end)

benchmark("table prealloc", function(i)
    local t = {nil}
    t[1] = i
end)
```

### 内存分析

```lua
local function memory_snapshot()
    return collectgarbage("count")
end

local function check_memory(label)
    local mem = collectgarbage("count")
    print(string.format("[%s] Memory: %.2f KB", label, mem))
    return mem
end

check_memory("before")
local data = {}
for i = 1, 100000 do
    data[i] = {id = i, value = math.random()}
end
check_memory("after alloc")
data = nil
collectgarbage("collect")
check_memory("after GC")
```
