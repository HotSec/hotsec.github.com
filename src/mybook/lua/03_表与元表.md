# Lua 表与元表

---

## 一、Table 基础

### 1.1 创建 Table

```lua
-- 空表
local t = {}

-- 数组
local arr = {1, 2, 3, 4, 5}

-- 哈希表
local person = {
    name = "Alice",
    age = 25,
    email = "alice@example.com",
}

-- 混合
local mixed = {
    "first",        -- [1]
    "second",       -- [2]
    name = "Alice", -- key = "name"
    [10] = "ten",   -- key = 10
}
```

### 1.2 访问与修改

```lua
local t = {name = "Alice", age = 25}

-- 两种访问方式
print(t.name)       -- "Alice"
print(t["name"])    -- "Alice"

-- 修改
t.name = "Bob"
t["age"] = 30

-- 新增
t.email = "bob@example.com"

-- 删除
t.email = nil
```

### 1.3 遍历

```lua
local t = {name = "Alice", age = 25, city = "Beijing"}

-- pairs: 遍历所有键值对（无序）
for k, v in pairs(t) do
    print(k, v)
end

-- ipairs: 遍历连续整数键（从1开始）
local arr = {10, 20, 30, 40}
for i, v in ipairs(arr) do
    print(i, v)  -- 1,10  2,20  3,30  4,40
end

-- # 长度（只计算连续整数键）
print(#arr)  -- 4
```

## 二、Metatable (元表)

### 2.1 概念

- 每个表可以关联一个元表 (metatable)
- 元表定义了表在特定操作下的行为
- 通过元方法 (metamethod) 实现运算符重载等

```lua
local t = {}
local mt = {}
setmetatable(t, mt)

-- 或
local t = setmetatable({}, mt)

-- 获取元表
getmetatable(t)  -- mt
```

### 2.2 算术元方法

```lua
local Vector = {}
Vector.__index = Vector

function Vector.new(x, y)
    return setmetatable({x = x, y = y}, Vector)
end

-- 加法
function Vector.__add(a, b)
    return Vector.new(a.x + b.x, a.y + b.y)
end

-- 减法
function Vector.__sub(a, b)
    return Vector.new(a.x - b.x, a.y - b.y)
end

-- 乘法（数乘）
function Vector.__mul(a, b)
    if type(a) == "number" then
        return Vector.new(a * b.x, a * b.y)
    elseif type(b) == "number" then
        return Vector.new(a.x * b, a.y * b)
    end
end

-- 相等
function Vector.__eq(a, b)
    return a.x == b.x and a.y == b.y
end

-- 字符串表示
function Vector.__tostring(v)
    return string.format("(%g, %g)", v.x, v.y)
end

local v1 = Vector.new(1, 2)
local v2 = Vector.new(3, 4)
local v3 = v1 + v2       -- (4, 6)
local v4 = v1 * 3        -- (3, 6)
print(v3)                 -- (4, 6)
```

### 2.3 所有元方法

| 元方法 | 触发操作 |
|--------|----------|
| `__add` | `a + b` |
| `__sub` | `a - b` |
| `__mul` | `a * b` |
| `__div` | `a / b` |
| `__mod` | `a % b` |
| `__pow` | `a ^ b` |
| `__unm` | `-a` |
| `__concat` | `a .. b` |
| `__eq` | `a == b` |
| `__lt` | `a < b` |
| `__le` | `a <= b` |
| `__index` | 访问不存在的键 |
| `__newindex` | 设置不存在的键 |
| `__call` | `table()` |
| `__tostring` | `tostring(table)` |
| `__len` | `#table` |
| `__pairs` | `pairs(table)` |
| `__ipairs` | `ipairs(table)` |

## 三、__index 与类实现

### 3.1 __index 查找链

```lua
-- 访问 t.key 时:
-- 1. 检查 t 中是否有 key
-- 2. 如果没有，检查 t 的元表是否有 __index
-- 3. 如果 __index 是表，在该表中查找 key
-- 4. 如果 __index 是函数，调用 __index(t, key)
```

### 3.2 类实现

```lua
local Animal = {}
Animal.__index = Animal

function Animal.new(name, sound)
    local instance = setmetatable({}, Animal)
    instance.name = name
    instance.sound = sound
    return instance
end

function Animal:speak()
    print(self.name .. " says " .. self.sound)
end

-- 继承
local Dog = setmetatable({}, {__index = Animal})
Dog.__index = Dog

function Dog.new(name, breed)
    local instance = Animal.new(name, "Woof")
    setmetatable(instance, Dog)
    instance.breed = breed
    return instance
end

function Dog:introduce()
    print("I'm " .. self.name .. ", a " .. self.breed)
end

local d = Dog.new("Rex", "Shepherd")
d:speak()      -- Rex says Woof (继承自 Animal)
d:introduce()  -- I'm Rex, a Shepherd
```

## 四、__newindex 与代理表

```lua
-- 只读表
function readonly(t)
    local proxy = {}
    setmetatable(proxy, {
        __index = t,
        __newindex = function(t, k, v)
            error("attempt to modify read-only table")
        end
    })
    return proxy
end

local config = readonly({host = "localhost", port = 8080})
print(config.host)  -- "localhost"
-- config.host = "other"  -- Error!

-- 日志代理
function logged(t)
    local proxy = {}
    setmetatable(proxy, {
        __index = function(_, k)
            print("accessing: " .. k)
            return t[k]
        end,
        __newindex = function(_, k, v)
            print("setting: " .. k .. " = " .. tostring(v))
            t[k] = v
        end,
    })
    return proxy
end
```

## 五、__call

```lua
local Counter = {}
Counter.__index = Counter

function Counter.new(start)
    local self = setmetatable({}, Counter)
    self.count = start or 0
    return self
end

function Counter.__call(self, increment)
    self.count = self.count + (increment or 1)
    return self.count
end

local c = Counter.new(0)
print(c())   -- 1
print(c())   -- 2
print(c(5))  -- 7
```
