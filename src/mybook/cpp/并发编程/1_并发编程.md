## 一、线程管理

### 1.1 std::thread

```cpp
#include <thread>

void hello(int id) {
    std::cout << "Hello from thread " << id << "\n";
}

std::thread t1(hello, 1);
std::thread t2([]() { std::cout << "Lambda thread\n"; });

t1.join();  // 等待完成
t2.join();  // 等待完成

// detach：分离线程
std::thread t3(hello, 3);
t3.detach(); // 后台运行，不可 join
```

### 1.2 线程管理 RAII

```cpp
class ThreadGuard {
    std::thread& t_;
public:
    explicit ThreadGuard(std::thread& t) : t_(t) {}
    ~ThreadGuard() {
        if (t_.joinable()) {
            t_.join();
        }
    }
    ThreadGuard(const ThreadGuard&) = delete;
    ThreadGuard& operator=(const ThreadGuard&) = delete;
};

void func() {
    std::thread t(hello, 1);
    ThreadGuard g(t);
    // 函数退出时自动 join
}
```

### 1.3 线程池

```cpp
class ThreadPool {
public:
    explicit ThreadPool(size_t num_threads) : stop_(false) {
        for (size_t i = 0; i < num_threads; ++i) {
            workers_.emplace_back([this] {
                while (true) {
                    std::function<void()> task;
                    {
                        std::unique_lock lock(mutex_);
                        cv_.wait(lock, [this] {
                            return stop_ || !tasks_.empty();
                        });
                        if (stop_ && tasks_.empty()) return;
                        task = std::move(tasks_.front());
                        tasks_.pop();
                    }
                    task();
                }
            });
        }
    }

    ~ThreadPool() {
        {
            std::unique_lock lock(mutex_);
            stop_ = true;
        }
        cv_.notify_all();
        for (auto& w : workers_) w.join();
    }

    template<typename F, typename... Args>
    auto submit(F&& f, Args&&... args) -> std::future<decltype(f(args...))> {
        using ReturnType = decltype(f(args...));
        auto task = std::make_shared<std::packaged_task<ReturnType()>>(
            std::bind(std::forward<F>(f), std::forward<Args>(args)...)
        );
        auto result = task->get_future();
        {
            std::unique_lock lock(mutex_);
            tasks_.emplace([task]() { (*task)(); });
        }
        cv_.notify_one();
        return result;
    }

private:
    std::vector<std::thread> workers_;
    std::queue<std::function<void()>> tasks_;
    std::mutex mutex_;
    std::condition_variable cv_;
    bool stop_;
};
```

## 二、同步原语

### 2.1 std::mutex

```cpp
std::mutex mtx;
int counter = 0;

void increment() {
    std::lock_guard<std::mutex> lock(mtx);
    counter++;
}

// recursive_mutex：同一线程可多次加锁
std::recursive_mutex rmtx;
void recursive_func(int n) {
    std::lock_guard<std::recursive_mutex> lock(rmtx);
    if (n > 0) recursive_func(n - 1);
}

// timed_mutex：超时加锁
std::timed_mutex tmtx;
if (tmtx.try_lock_for(std::chrono::milliseconds(100))) {
    // 加锁成功
    tmtx.unlock();
}
```

### 2.2 std::shared_mutex (读写锁)

```cpp
std::shared_mutex rwmtx;
std::map<std::string, int> cache;

int read(const std::string& key) {
    std::shared_lock lock(rwmtx); // 共享读锁
    return cache.at(key);
}

void write(const std::string& key, int value) {
    std::unique_lock lock(rwmtx); // 独占写锁
    cache[key] = value;
}
```

### 2.3 std::condition_variable

```cpp
std::mutex mtx;
std::condition_variable cv;
bool ready = false;

void worker() {
    std::unique_lock lock(mtx);
    cv.wait(lock, [] { return ready; }); // 等待条件满足
    // 执行工作
}

void notifier() {
    {
        std::lock_guard lock(mtx);
        ready = true;
    }
    cv.notify_one(); // 或 cv.notify_all()
}
```

### 2.4 std::latch / std::barrier (C++20)

```cpp
#include <latch>
#include <barrier>

// latch：一次性同步点
std::latch done(3);

void task() {
    doWork();
    done.count_down(); // 计数 -1
}

done.wait(); // 等待计数归零

// barrier：可重用同步点
std::barrier sync_point(3, []() noexcept {
    std::cout << "All threads reached barrier\n";
});

void parallel_task() {
    phase1();
    sync_point.arrive_and_wait(); // 等待所有线程
    phase2();
    sync_point.arrive_and_wait();
}
```

### 2.5 std::counting_semaphore (C++20)

```cpp
#include <semaphore>

// 限制并发数
std::counting_semaphore<5> sem(5); // 最多5个并发

void limited_task() {
    sem.acquire();
    doWork();
    sem.release();
}
```

## 三、原子操作与内存模型

### 3.1 std::atomic

```cpp
std::atomic<int> counter(0);

counter++;                     // 原子自增
counter.fetch_add(1);          // 原子加
counter.load();                // 原子读
counter.store(42);             // 原子写
counter.exchange(0);           // 原子交换，返回旧值

// CAS (Compare-And-Swap)
int expected = 42;
bool success = counter.compare_exchange_strong(expected, 100);
// 如果 counter == expected，则 counter = 100，返回 true
// 否则 expected = counter，返回 false
```

### 3.2 内存序 (Memory Order)

```cpp
// memory_order_relaxed：无顺序保证
counter.fetch_add(1, std::memory_order_relaxed);

// memory_order_acquire：后续读写不能重排到此之前
auto val = counter.load(std::memory_order_acquire);

// memory_order_release：之前的读写不能重排到此之后
counter.store(1, std::memory_order_release);

// memory_order_acq_rel：acquire + release
counter.fetch_add(1, std::memory_order_acq_rel);

// memory_order_seq_cst：顺序一致性（默认，最严格）
counter.load(std::memory_order_seq_cst);
```

### 3.3 内存序示例

```cpp
std::atomic<bool> ready(false);
int data = 0;

// 生产者
void producer() {
    data = 42;                                    // 写数据
    ready.store(true, std::memory_order_release); // 释放
}

// 消费者
void consumer() {
    while (!ready.load(std::memory_order_acquire)) { // 获取
        // 自旋等待
    }
    // 一定能看到 data = 42
    assert(data == 42);
}
```

| 内存序 | 保证 | 性能 |
|--------|------|------|
| relaxed | 原子性 | 最快 |
| acquire/release | 单线程 happens-before | 较快 |
| acq_rel | 双向同步 | 中等 |
| seq_cst | 全局顺序 | 最慢 |

## 四、Futures & Promises

### 4.1 std::future / std::promise

```cpp
std::promise<int> prom;
std::future<int> fut = prom.get_future();

std::thread t([&prom] {
    int result = heavyComputation();
    prom.set_value(result);
});

int result = fut.get(); // 阻塞等待
t.join();
```

### 4.2 std::async

```cpp
// 异步启动策略
auto fut1 = std::async(std::launch::async, [] {
    return heavyComputation(); // 新线程
});

auto fut2 = std::async(std::launch::deferred, [] {
    return lazyComputation(); // get() 时才执行
});

int result = fut1.get();
```

### 4.3 std::packaged_task

```cpp
std::packaged_task<int(int, int)> task([](int a, int b) {
    return a + b;
});

std::future<int> fut = task.get_future();
std::thread t(std::move(task), 3, 5);

int result = fut.get(); // 8
t.join();
```

## 五、无锁数据结构

### 5.1 无锁栈

```cpp
template<typename T>
class LockFreeStack {
    struct Node {
        T data;
        Node* next;
        Node(T val) : data(std::move(val)), next(nullptr) {}
    };

    std::atomic<Node*> head_;

public:
    void push(T value) {
        Node* new_node = new Node(std::move(value));
        new_node->next = head_.load(std::memory_order_relaxed);
        while (!head_.compare_exchange_weak(
            new_node->next, new_node,
            std::memory_order_release,
            std::memory_order_relaxed)) {}
    }
};
```
