# 测试

## 一、单元测试

- 测试文件以 `_test.go` 结尾
- 测试函数以 `Test` 开头，参数为 `*testing.T`
- `go test -v`

```go
func TestAdd(t *testing.T) {
    result := Add(1, 2)
    if result != 3 {
        t.Errorf("Add(1, 2) = %d; want 3", result)
    }
}
```

### 1.1 子测试

```go
func TestSplit(t *testing.T) {
    t.Run("simple", func(t *testing.T) {
        s := strings.Split("a,b,c", ",")
        if len(s) != 3 {
            t.Errorf("expected 3, got %d", len(s))
        }
    })

    t.Run("empty", func(t *testing.T) {
        s := strings.Split("", ",")
        if len(s) != 1 {
            t.Errorf("expected 1, got %d", len(s))
        }
    })
}
```

### 1.2 表驱动测试

```go
func TestAdd(t *testing.T) {
    tests := []struct {
        name     string
        a, b     int
        expected int
    }{
        {"positive", 1, 2, 3},
        {"negative", -1, -2, -3},
        {"zero", 0, 0, 0},
        {"mixed", 1, -1, 0},
    }

    for _, tt := range tests {
        t.Run(tt.name, func(t *testing.T) {
            result := Add(tt.a, tt.b)
            if result != tt.expected {
                t.Errorf("Add(%d, %d) = %d; want %d", tt.a, tt.b, result, tt.expected)
            }
        })
    }
}
```

## 二、基准测试

```go
func BenchmarkAdd(b *testing.B) {
    for i := 0; i < b.N; i++ {

    }

```

```bash

```

## 三、模糊测试

```go
func FuzzAdd(f *testing.F) {
    f.Add(1, 2)
    f.Fuzz(func(t *testing.T, a, b int) {
        result := Add(a, b)
        if result != a+b {
            t.Errorf("Add(%d, %d) = %d", a, b, result)
        }
    })
}
```

```bash
go test -fuzz=FuzzAdd -fuzztime=30s
```

## 四、Mock 接口

### 4.1 手动 Mock

```go
type UserRepository interface {
    GetByID(ctx context.Context, id int) (*User, error)
    Save(ctx context.Context, user *User) error
}

type MockUserRepository struct {
    GetByIDFunc func(ctx context.Context, id int) (*User, error)
    SaveFunc    func(ctx context.Context, user *User) error
}

func (m *MockUserRepository) GetByID(ctx context.Context, id int) (*User, error) {
    return m.GetByIDFunc(ctx, id)
}

func (m *MockUserRepository) Save(ctx context.Context, user *User) error {
    return m.SaveFunc(ctx, user)
}

func TestGetUser(t *testing.T) {
    repo := &MockUserRepository{
        GetByIDFunc: func(ctx context.Context, id int) (*User, error) {
            return &User{ID: id, Name: "test"}, nil
        },
    }

    user, err := GetUser(context.Background(), repo, 1)
    if err != nil {
        t.Fatal(err)
    }
    if user.Name != "test" {
        t.Errorf("expected test, got %s", user.Name)
    }
}
```

func TestWithPatch(t *testing.T) {
    patches := gomonkey.ApplyMethod(time.Now, func() time.Time {
        return time.Date(2024, 1, 1, 0, 0, 0, 0, time.UTC)

        })

## 七、测试 MySQL 和 Redis

```go
func TestWithMySQL(t *testing.T) {
    db, mock, err := sqlmock.New()
    if err != nil {
        t.Fatal(err)
    }
    defer db.Close()

    mock.ExpectQuery("SELECT (.+) FROM users").
        WithArgs(1).
        WillReturnRows(sqlmock.NewRows([]string{"id", "name"}).AddRow(1, "Go"))

    var user User
    db.QueryRow("SELECT id, name FROM users WHERE id = ?", 1).Scan(&user.ID, &user.Name)
}

func TestWithRedis(t *testing.T) {
    rdb, mock := redismock.NewClientMock()
    mock.ExpectGet("key").SetVal("value")

    val, err := rdb.Get(context.Background(), "key").Result()
    if err != nil {
        t.Fatal(err)
    }
    if val != "value" {
        t.Errorf("expected value, got %s", val)
    }
}
```
