# Go 调度器深入

> 主要参考：《Go语言高级编程（第2版）》第4.2节 — 柴树杉、曹春晖 著

Go 调度器是 Go 运行时的核心组件，负责将 Goroutine（G）调度到操作系统线程（M）上执行。本文深入分析调度器的实现原理、调度策略和关键数据结构。

---

## 一、GMP 模型详解

### 1.1 核心数据结构

#### G (Goroutine)

```go
type g struct {
    stack       stack   // 栈范围 [lo, hi)
    stackguard0 uintptr // 栈溢出检查
    stackguard1 uintptr // 栈溢出检查（C 栈）

    m              *m      // 当前绑定的 M
    sched          gobuf   // 调度上下文
    atomicstatus   uint32  // 状态
    goid           int64   // goroutine ID
    waitsince      int64   // 等待开始时间
    waitreason     waitReason // 等待原因
    preempt        bool    // 抢占标志
    lockedm        *m      // 锁定的 M
    timer          *timer  // 时间器
}

type gobuf struct {
    sp   uintptr // 栈指针
    pc   uintptr // 程序计数器
    g    guintptr
    ret  uintptr // 返回值
}

type stack struct {
    lo uintptr // 栈底
    hi uintptr // 栈顶
}
```

G 的状态：

```
_Gidle       = iota // 0: 刚分配，尚未初始化
_Grunnable          // 1: 在运行队列中等待
_Grunning           // 2: 正在执行
_Gsyscall           // 3: 系统调用中
_Gwaiting           // 4: 等待中（channel/锁/IO等）
_Gdead              // 6: 已经退出或被回收
_Gcopystack         // 8: 栈正在拷贝
_Gpreempted         // 9: 被抢占
```

## 二、调度流程

### 2.1 创建 Goroutine

```go
//go:nosplit
func newproc(siz int32, fn *funcval) {
    argp := add(unsafe.Pointer(&fn), sys.PtrSize)
    gp := getg()
    pc := getcallerpc()
    
    systemstack(func() {
        newg := newproc1(fn, argp, siz, gp, pc)
        
        // 放入运行队列
        _p_ := getg().m.p.ptr()
        runqput(_p_, newg, true)
        
        // 唤醒或创建 M
        if mainStarted {
            wakep()
        }
    })
}

func newproc1(fn *funcval, argp unsafe.Pointer, narg int32, callergp *g, callerpc uintptr) *g {
    _g_ := getg()
    _p_ := _g_.m.p.ptr()
    
    // 从 P 的空闲列表获取 G
    newg := gfget(_p_)
    if newg == nil {
        // 创建新的 G
        newg = malg(_StackMin)
        casgstatus(newg, _Gidle, _Gdead)
        allgadd(newg)
    }
    
    // 设置栈和调度信息
    totalSize := uintptr(4*goarch.PtrSize + uintptr(narg))
    sp := newg.stack.hi - totalSize
    memmove(unsafe.Pointer(sp), argp, uintptr(narg))
    
    // 设置调度上下文
    newg.sched.sp = sp
    newg.sched.pc = abi.FuncPCABI0(goexit) + sys.PCQuantum
    newg.sched.g = guintptr(unsafe.Pointer(newg))
    
    // 设置入口函数
    gostartcallfn(&newg.sched, fn)
    
    casgstatus(newg, _Gdead, _Grunnable)
    newg.goid = _p_.goidcache
    _p_.goidcache++
    
    return newg
}
```

### 2.2 调度循环

```go
func schedule() {
    _g_ := getg()
    
    // 检查是否需要 GC
    if sched.gcwaiting != 0 {
        gcstopm()
        return
    }
    
    var gp *g
    var inheritTime bool
    
    // 1. 检查是否需要 GC 辅助
    if trace.enabled || trace.shutdown {
        gp = traceReader()
    }
    
    // 2. 检查是否有 GC 工作
    if gp == nil && gcBlackenEnabled != 0 {
        gp = gcController.findRunnableGCWorker(_g_.m.p.ptr())
    }
    
    // 3. 每 61 次检查全局队列
    if gp == nil {
        if _g_.m.p.ptr().schedtick%61 == 0 && sched.runqsize > 0 {
            lock(&sched.lock)
            gp = globrunqget(_g_.m.p.ptr(), 1)
            unlock(&sched.lock)
        }
    }
    
    // 4. 从本地队列获取
    if gp == nil {
        gp, inheritTime = runqget(_g_.m.p.ptr())
    }
    
    // 5. 从全局队列获取
    if gp == nil {
        if gp = globrunqget(_g_.m.p.ptr(), 0); gp == nil {
            // 6. 从网络轮询器获取
            if netpollinited && netpollWaiters.Load() > 0 && sched.lastpoll.Load() != 0 {
                if list := netpoll(0); !list.empty() {
                    gp = list.pop()
                    injectglist(&list)
                }
            }
        }
    }
    
    // 7. 工作窃取
    if gp == nil {
        gp, inheritTime = findrunnable()
    }
    
    // 执行 G
    execute(gp, inheritTime)
}

func execute(gp *g, inheritTime bool) {
    _g_ := getg()
    
    // 绑定 G 到 M
    _g_.m.curg = gp
    gp.m = _g_.m
    
    // 状态转换
    casgstatus(gp, _Grunnable, _Grunning)
    gp.waitsince = 0
    gp.preempt = false
    gp.stackguard0 = gp.stack.lo + stackGuard
    
    // 切换到 G 执行
    gogo(&gp.sched)
}
```

### 2.3 Goroutine 切换

```go
// gogo 由汇编实现
// 从 gobuf 恢复上下文并跳转到 G 执行
TEXT gogo(SB), NOSPLIT, $0-8
    MOVQ buf+0(FP), BX     // gobuf
    MOVQ gobuf_g(BX), DX   // G
    MOVQ 0(DX), CX         // 确保G不为nil
    get_tls(CX)
    MOVQ DX, g(CX)         // 设置 TLS 中的 G
    MOVQ gobuf_sp(BX), SP  // 恢复 SP
    MOVQ gobuf_ret(BX), AX
    MOVQ gobuf_ctxt(BX), DX
    MOVQ gobuf_bp(BX), BP
    MOVQ $0, gobuf_sp(BX)  // 清空
    MOVQ $0, gobuf_ret(BX)
    MOVQ $0, gobuf_ctxt(BX)
    MOVQ $0, gobuf_bp(BX)
    MOVQ gobuf_pc(BX), BX  // 跳转地址
    JMP BX
```

### 2.4 Goroutine 挂起

```go
func gopark(unlockf func(*g, unsafe.Pointer) bool, lock unsafe.Pointer, reason waitReason, traceReason traceBlockReason, traceskip int) {
    mp := acquirem()
    gp := mp.curg
    
    // 设置状态
    casgstatus(gp, _Grunning, _Gwaiting)
    gp.waitreason = reason
    
    // 调用解锁函数
    if !unlockf(gp, lock) {
        casgstatus(gp, _Gwaiting, _Grunning)
        releasem(mp)
        return
    }
    
    // 切换到 g0 执行调度
    mcall(park_m)
}

func park_m(gp *g) {
    _g_ := getg()
    
    casgstatus(gp, _Grunning, _Gwaiting)
    dropg()
    
    // 重新调度
    schedule()
}
```

```go
func ready(gp *g, traceskip int, next bool) {
```

## 三、工作窃取

### 3.1 窃取算法

```go
func findrunnable() (gp *g, inheritTime bool) {
    _g_ := getg()
    _p_ := _g_.m.p.ptr()
    
    // 1. 本地队列
    if gp, inheritTime = runqget(_p_); gp != nil {
        return
    }
    
    // 2. 全局队列
    if sched.runqsize > 0 {
        lock(&sched.lock)
        gp = globrunqget(_p_, 0)
        unlock(&sched.lock)
        if gp != nil {
            return gp, false
        }
    }
    
    // 3. 网络轮询器
    if netpollinited && netpollWaiters.Load() > 0 {
        if list := netpoll(0); !list.empty() {
            gp = list.pop()
            injectglist(&list)
            return gp, false
        }
    }
    
    // 4. 工作窃取
    procs := uint32(gomaxprocs)
    ranTimer := false
    
    for i := 0; i < 4; i++ {
        for enum := stealOrder.start(fastrand()); !enum.done(); enum.next() {
            if sched.gcwaiting != 0 {
                return nil, false
            }
            
            p2 := allp[enum.position()]
            if p2 == _p_ {
                continue
            }
            
            // 窃取本地队列
            if gp := runqsteal(_p_, p2, stealRunNextQ); gp != nil {
                return gp, false
            }
            
            // 窃取定时器
            if i > 2 && ranTimer {
                if gp := runqsteal(_p_, p2, stealRunNextQ); gp != nil {
                    return gp, false
                }
            }
        }
    }
    
    // 5. 休眠当前 M
    stopm()
    return nil, false
}
```

### 3.2 窃取实现

```go
func runqsteal(_p_, p2 *p, stealRunNextQ bool) *g {
    t := _p_.runqtail
    n := runqgrab(p2, &_p_.runq, t, stealRunNextQ)
    if n == 0 {
        return nil
    }
    n--
    gp := _p_.runq[(t+n)%uint32(len(_p_.runq))].ptr()
    if n == 0 {
        _p_.runqtail = t
    } else {
        _p_.runqtail = t + n
    }
    return gp
}

func runqgrab(_p_ *p, batch *gQueue, batchHead uint32, stealRunNextQ bool) uint32 {
    for {
        h := atomic.LoadAcq(&_p_.runqhead)
        t := atomic.LoadAcq(&_p_.runqtail)
        n := t - h
        n = n - n/2  // 窃取一半
        
        if n == 0 {
            if stealRunNextQ {
                if next := _p_.runnext; next != 0 {
                    if _p_.status == _Prunning {
                        return 0
                    }
                    if !atomic.CasRel(&_p_.runnext, next, 0) {
                        continue
                    }
                    batch.push(next.ptr())
                    return 1
                }
            }
            return 0
        }
        
        if n > uint32(len(_p_.runq)/2) {
            continue
        }
        
        for i := uint32(0); i < n; i++ {
            g := _p_.runq[(h+i)%uint32(len(_p_.runq))]
            batch.push(g.ptr())
        }
        
        if atomic.CasRel(&_p_.runqhead, h, h+n) {
            return n
        }
    }
}
```

## 四、系统调用处理

### 4.1 进入系统调用

```go
func entersyscall() {
    _g_ := getg()
    
    _g_.m.locks++
    
    // 保存调度上下文
    save(getcallerpc(), getcallersp())
    
    // 设置状态
    casgstatus(_g_, _Grunning, _Gsyscall)
    _g_.syscallsp = _g_.sched.sp
    _g_.syscallpc = _g_.sched.pc
    
    // 解绑 P（但保留引用）
    _g_.m.oldp.set(_g_.m.p.ptr())
    _g_.m.oldp.ptr().status = _Psyscall
    
    _g_.m.locks--
}

func entersyscallblock() {
    _g_ := getg()
    
    // 保存调度上下文
    save(getcallerpc(), getcallersp())
    
    // 设置状态
    casgstatus(_g_, _Grunning, _Gsyscall)
    
    // 解绑 P
    _p_ := releasep()
    handoffp(_p_)
}
```

### 4.2 退出系统调用

```go
func exitsyscall() {
    _g_ := getg()
    
    // 尝试重新绑定原来的 P
    oldp := _g_.m.oldp.ptr()
    _g_.m.oldp = 0
    
    if exitsyscallfast(oldp) {
        // 成功重新绑定
        casgstatus(_g_, _Gsyscall, _Grunning)
        return
    }
    
    // 需要重新调度
    mcall(exitsyscall0)
}

func exitsyscallfast(oldp *p) bool {
    _g_ := getg()
    
    // 1. 尝试绑定原来的 P
    if oldp != nil && oldp.status == _Psyscall && atomic.Cas(&oldp.status, _Psyscall, _Pidle) {
        wirep(oldp)
        return true
    }
    
    // 2. 尝试获取空闲 P
    if sched.pidle != 0 {
        lock(&sched.lock)
        if _p_ := pidleget(); _p_ != nil {
            unlock(&sched.lock)
            wirep(_p_)
            return true
        }
        unlock(&sched.lock)
    }
    
    return false
}

func exitsyscall0(gp *g) {
    casgstatus(gp, _Gsyscall, _Grunnable)
    dropg()
    
    // 放入全局队列
    lock(&sched.lock)
    globrunqput(gp)
    unlock(&sched.lock)
    
    // 停止当前 M
    stopm()
    
    schedule()
}
```

### 5.3 抢占时机

```go
// 1. 系统监控检测长时间运行的 G
func retake(now int64) uint32 {
    n := 0
    lock(&allpLock)
    for i := 0; i < len(allp); i++ {
        _p_ := allp[i]
        if _p_ == nil {
            continue
        }
        
        pd := &_p_.sysmontick
        s := _p_.status
        
        if s == _Prunning || s == _Psyscall {
            t := int64(_p_.schedtick)
            if int64(pd.schedtick) != t {
                pd.schedtick = uint32(t)
                pd.schedwhen = now
                continue
            }
            
            // 运行超过 10ms
            if now-pd.schedwhen > 10*1000*1000 {
                preemptone(_p_)
            }
        }
        
        if s == _Psyscall {
            // 系统调用超过 10ms
            if now-pd.syscallwhen > 10*1000*1000 {
                handoffp(_p_)
            }
        }
    }
    unlock(&allpLock)
    return uint32(n)
}
```

## 六、调度器调优

### 6.1 GOMAXPROCS

```go
// 设置 P 的数量
func GOMAXPROCS(n int) int {
    if n < 0 {
        return int(gomaxprocs)
    }
    
    lock(&sched.lock)
    ret := int(gomaxprocs)
    
    if n > _MaxGomaxprocs {
        n = _MaxGomaxprocs
    }
    
    if n != int(gomaxprocs) {
        lock(&allpLock)
        
        // 调整 P 的数量
        procresize(int32(n))
        
        unlock(&allpLock)
    }
    
    unlock(&sched.lock)
    return ret
}
```

### 6.2 LockOSThread

```go
// 绑定 G 到当前 M
func LockOSThread() {
    if atomic.Load(&go11os) == 0 {
        return
    }
    _g_ := getg()
    _g_.m.lockedExt++
    if _g_.m.lockedExt == 0 {
        _g_.m.lockedExt--
        panic("LockOSThread nesting overflow")
    }
    dolockOSThread()
}

func dolockOSThread() {
    _g_ := getg()
    _g_.m.lockedg.set(_g_)
    _g_.lockedm.set(_g_.m)
}

func UnlockOSThread() {
    if atomic.Load(&go11os) == 0 {
        return
    }
    _g_ := getg()
    if _g_.m.lockedExt == 0 {
        return
    }
    _g_.m.lockedExt--
    if _g_.m.lockedExt == 0 {
        dounlockOSThread()
    }
}
```

## 七、调试与分析

### 7.1 GODEBUG=schedtrace

字段解释：
- `gomaxprocs`: P 的数量
- `idleprocs`: 空闲 P 的数量
- `threads`: M 的数量
- `spinningthreads`: 正在自旋的 M 数量
- `idlethreads`: 空闲 M 的数量
- `runqueue`: 全局队列中 G 的数量
- `[...]`: 每个 P 的本地队列长度

### 7.2 runtime 调度器指标

```go
import "runtime"

func printSchedulerStats() {
    var buf [128]runtime.MemStats
    runtime.ReadMemStats(&buf[0])
    
    fmt.Printf("NumGoroutine: %d\n", runtime.NumGoroutine())
    fmt.Printf("NumCPU: %d\n", runtime.NumCPU())
    fmt.Printf("GOMAXPROCS: %d\n", runtime.GOMAXPROCS(0))
    fmt.Printf("NumCgoCall: %d\n", runtime.NumCgoCall())
}
```
