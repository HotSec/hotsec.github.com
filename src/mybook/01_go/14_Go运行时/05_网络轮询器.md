# Go 网络轮询器

> 主要参考：《Go语言高级编程（第2版）》第4.5节 — 柴树杉、曹春晖 著

Go 网络轮询器（Netpoller）是 Go 运行时实现高效网络 I/O 的核心组件。它基于 I/O 多路复用技术（epoll/kqueue/IOCP），将同步的网络操作转换为异步的非阻塞操作，使 Goroutine 能够在等待网络 I/O 时让出 CPU，从而实现高并发网络编程。

---

## 一、设计目标

### 1.1 问题背景

传统的阻塞 I/O 模型：

```go
// 传统阻塞 I/O
func handleConn(conn net.Conn) {
    buf := make([]byte, 1024)
    n, err := conn.Read(buf)  // 阻塞等待数据
    if err != nil {
        return
    }
    // 处理数据
}
```

问题：
- 每个连接需要一个线程
- 线程数受限于系统资源
- 线程切换开销大
- 难以支持大量并发连接

### 1.2 Go 的解决方案

Go 通过网络轮询器实现非阻塞 I/O：

```
┌─────────────────────────────────────────────────────────────┐
│                  Go 网络模型                                 │
├─────────────────────────────────────────────────────────────┤
│                                                             │
│  Goroutine 调用 Read()                                      │
│       │                                                     │
│       ▼                                                     │
│  非阻塞读取（syscall.Read）                                 │
│       │                                                     │
│       ├── 有数据 → 直接返回                                 │
│       │                                                     │
│       └── 无数据（EAGAIN）                                  │
│              │                                              │
│              ▼                                              │
│           挂起 Goroutine                                    │
│              │                                              │
│              ▼                                              │
│           将 fd 注册到 epoll                                │
│              │                                              │
│              ▼                                              │
│           M 继续执行其他 G                                  │
│              │                                              │
│              ▼                                              │
│           数据就绪 → epoll_wait 返回                        │
│              │                                              │
│              ▼                                              │
│           唤醒 Goroutine                                   │
│                                                             │
└─────────────────────────────────────────────────────────────┘
```

## 二、核心数据结构

### 2.1 pollDesc

pollDesc 描述一个被轮询的文件描述符：

```go
type pollDesc struct {
    link *pollDesc // 链表下一个
    
    fd      uintptr    // 文件描述符
    closing bool       // 是否关闭
    
    // 读操作相关
    rg     uintptr     // 等待读的 goroutine
    rd     int64       // 读截止时间
    
    // 写操作相关
    wg     uintptr     // 等待写的 goroutine
    wd     int64       // 写截止时间
    
    // 用户数据
    ctx    unsafe.Pointer
}

const (
    pdReady uintptr = 1  // 就绪状态
    pdWait  uintptr = 2  // 等待状态
)
```

pollCache 管理 pollDesc 的缓存：

```go
type pollCache struct {
    lock  mutex
    first *pollDesc
    // pollDesc 缓存池
}
```

## 三、工作流程

### 3.1 初始化

```go
var (
    netpollInited uint32
    epfd          int32  // epoll fd
)

func netpollGenericInit() {
    if atomic.Cas(&netpollInited, 0, 1) {
        // 创建 epoll 实例
        epfd = epollcreate1(_EPOLL_CLOEXEC)
        
        // 创建 pipe 用于唤醒
        r, w, _ := nonblockingPipe()
        epollctl(epfd, _EPOLL_CTL_ADD, r, &epollevent{
            events: _EPOLLIN,
            data:   (uintptr(unsafe.Pointer(&netpollBreakRd))),
        })
        
        netpollBreakRd = uintptr(r)
        netpollBreakWr = uintptr(w)
    }
}
```

### 3.2 注册文件描述符

```go
func (pd *pollDesc) init(fd uintptr) error {
    // 初始化 pollDesc
    pd.fd = fd
    pd.rg = 0
    pd.wg = 0
    
    // 注册到 epoll
    var ev epollevent
    ev.events = _EPOLLIN | _EPOLLOUT | _EPOLLRDHUP | _EPOLLET
    *(**pollDesc)(unsafe.Pointer(&ev.data)) = pd
    
    if epollctl(epfd, _EPOLL_CTL_ADD, int32(fd), &ev) != 0 {
        return errno
    }
    
    return nil
}
```

### 3.3 等待事件

```go
func netpoll(delay int64) gList {
    var list gList
    
    // 计算超时时间
    var timeout int32
    if delay < 0 {
        timeout = -1
    } else if delay == 0 {
        timeout = 0
    } else {
        timeout = int32(delay / 1000000)
    }
    
    // 调用 epoll_wait
    var events [128]epollevent
    n := epollwait(epfd, &events[0], int32(len(events)), timeout)
    
    if n < 0 {
        return list
    }
    
    // 处理就绪事件
    for i := 0; i < int(n); i++ {
        ev := &events[i]
        
        // 获取 pollDesc
        pd := *(**pollDesc)(unsafe.Pointer(&ev.data))
        
        // 检查事件类型
        var mode int32
        if ev.events&(_EPOLLIN|_EPOLLRDHUP|_EPOLLHUP|_EPOLLERR) != 0 {
            mode += 'r'
        }
        if ev.events&(_EPOLLOUT|_EPOLLHUP|_EPOLLERR) != 0 {
            mode += 'w'
        }
        
        // 唤醒等待的 goroutine
        if mode != 0 {
            netpollready(&list, pd, mode)
        }
    }
    
    return list
}

func netpollready(list *gList, pd *pollDesc, mode int32) {
    var rg, wg *g
    
    if mode == 'r' || mode == 'r'+'w' {
        rg = netpollunblock(pd, 'r')
    }
    if mode == 'w' || mode == 'r'+'w' {
        wg = netpollunblock(pd, 'w')
    }
    
    if rg != nil {
        list.push(rg)
    }
    if wg != nil {
        list.push(wg)
    }
}
```

### 3.4 挂起 Goroutine

```go
func netpollblock(pd *pollDesc, mode int32, waitio bool) bool {
    gpp := &pd.rg
    if mode == 'w' {
        gpp = &pd.wg
    }
    
    // 设置为等待状态
    for {
        old := *gpp
        if old == pdReady {
            *gpp = 0
            return true
        }
        if old != 0 {
            throw("netpollblock: double wait")
        }
        if atomic.Casuintptr(gpp, 0, pdWait) {
            break
        }
    }
    
    // 挂起 goroutine
    if waitio {
        gopark(netpollblockcommit, unsafe.Pointer(gpp), waitReasonIOWait, traceBlockNet, 0)
    }
    
    // 被唤醒后检查状态
    old := atomic.Xchguintptr(gpp, 0)
    return old == pdReady
}

func netpollblockcommit(gp *g, gpp unsafe.Pointer) bool {
    return atomic.Casuintptr((*uintptr)(gpp), pdWait, uintptr(unsafe.Pointer(gp)))
}
```

### 3.5 唤醒 Goroutine

```go
func netpollunblock(pd *pollDesc, mode int32) *g {
    gpp := &pd.rg
    if mode == 'w' {
        gpp = &pd.wg
    }
    
    for {
        old := *gpp
        if old == pdReady {
            return nil
        }
        if old == 0 {
            *gpp = pdReady
            return nil
        }
        if atomic.Casuintptr(gpp, old, pdReady) {
            return (*g)(unsafe.Pointer(old))
        }
    }
}
```

## 五、网络操作实现

### 5.1 netFD 结构

```go
type netFD struct {
    pfd poll.FD
    
    // 网络 type
    family      int
    sotype      int
    isConnected bool
    net         string
    laddr       Addr
    raddr       Addr
}

type FD struct {
    fdmu fdMutex
    
    Sysfd       int
    ZeroReadIsEOF bool
    
    // pollDesc
    pd pollDesc
}
```

### 5.2 Read 实现

```go
func (fd *FD) Read(p []byte) (int, error) {
    if err := fd.readLock(); err != nil {
        return 0, err
    }
    defer fd.readUnlock()
    
    for {
        // 非阻塞读取
        n, err := syscall.Read(fd.Sysfd, p)
        if err != nil {
            n = 0
            
            // 数据未就绪
            if err == syscall.EAGAIN {
                // 等待数据
                if err = fd.pd.waitRead(); err == nil {
                    continue
                }
            }
        }
        
        return n, err
    }
}

func (pd *pollDesc) waitRead() error {
    return pd.wait('r', false)
}

func (pd *pollDesc) wait(mode int32, waitio bool) error {
    // 挂起 goroutine
    if !netpollblock(pd, mode, waitio) {
        return errClosing
    }
    return nil
}
```

### 5.3 Write 实现

```go
func (fd *FD) Write(p []byte) (int, error) {
    if err := fd.writeLock(); err != nil {
        return 0, err
    }
    defer fd.writeUnlock()
    
    var nn int
    for {
        // 非阻塞写入
        n, err := syscall.Write(fd.Sysfd, p[nn:])
        if n > 0 {
            nn += n
        }
        if nn == len(p) {
            return nn, err
        }
        
        if err == syscall.EAGAIN {
            // 等待可写
            if err = fd.pd.waitWrite(); err == nil {
                continue
            }
        }
        
        return nn, err
    }
}
```

### 5.4 Accept 实现

```go
func (fd *FD) Accept() (int, syscall.Sockaddr, string, error) {
    for {
        // 非阻塞 accept
        n, sa, err := syscall.Accept(fd.Sysfd)
        if err == nil {
            return n, sa, "", nil
        }
        
        switch err {
        case syscall.EAGAIN:
            // 等待新连接
            if err = fd.pd.waitRead(); err == nil {
                continue
            }
        case syscall.ECONNABORTED:
            continue
        }
        
        return -1, nil, "accept", err
    }
}
```

## 六、定时器与截止时间

### 6.1 设置截止时间

```go
func (fd *FD) SetDeadline(t time.Time) error {
    return fd.setDeadline(t, 'r'+'w')
}

func (fd *FD) SetReadDeadline(t time.Time) error {
    return fd.setDeadline(t, 'r')
}

func (fd *FD) SetWriteDeadline(t time.Time) error {
    return fd.setDeadline(t, 'w')
}

func (fd *FD) setDeadline(t time.Time, mode int) error {
    var d int64
    if !t.IsZero() {
        d = int64(time.Until(t))
        if d <= 0 {
            d = -1
        }
    }
    
    // 设置截止时间
    if mode == 'r' || mode == 'r'+'w' {
        fd.pd.rd = d
    }
    if mode == 'w' || mode == 'r'+'w' {
        fd.pd.wd = d
    }
    
    // 唤醒等待的 goroutine
    netpollunblock(&fd.pd, int32(mode))
    
    return nil
}
```

### 6.2 超时处理

```go
func netpollblock(pd *pollDesc, mode int32, waitio bool) bool {
    // ...
    
    // 计算等待时间
    var waitDeadline int64
    if mode == 'r' {
        waitDeadline = pd.rd
    } else {
        waitDeadline = pd.wd
    }
    
    // 设置定时器
    if waitDeadline > 0 {
        timer := time.NewTimer(time.Duration(waitDeadline))
        defer timer.Stop()
        
        select {
        case <-timer.C:
            // 超时
            return false
        case <-ch:
            // 数据就绪
            return true
        }
    }
    
    // ...
}
```

### 7.2 macOS/BSD (kqueue)

```go
func netpollopen(fd uintptr, pd *pollDesc) int32 {
    var kev [2]kevent
    kev[0].ident = uint32(fd)
    kev[0].filter = _EVFILT_READ
    kev[0].flags = _EV_ADD | _EV_ENABLE | _EV_CLEAR
    kev[0].udata = (*byte)(unsafe.Pointer(pd))
    
    kev[1].ident = uint32(fd)
    kev[1].filter = _EVFILT_WRITE
    kev[1].flags = _EV_ADD | _EV_ENABLE | _EV_CLEAR
    kev[1].udata = (*byte)(unsafe.Pointer(pd))
    
    return kevent(kq, &kev[0], 2, nil, 0, nil)
}

func netpoll(delay int64) gList {
    var events [128]kevent
    n := kevent(kq, nil, 0, &events[0], int32(len(events)), &timeout)
    // ...
}
```

## 八、性能优化

### 8.1 边缘触发 vs 水平触发

Go 使用边缘触发（ET）模式：

```go
// 边缘触发：只在状态变化时通知
ev.events = _EPOLLIN | _EPOLLOUT | _EPOLLET

// 优点：减少事件通知次数
// 缺点：需要一次性读完/写完所有数据
```

### 8.2 非阻塞 I/O

```go
// 设置非阻塞
func setNonblock(fd int) error {
    flags, err := syscall.FcntlInt(uintptr(fd), syscall.F_GETFL, 0)
    if err != nil {
        return err
    }
    _, err = syscall.FcntlInt(uintptr(fd), syscall.F_SETFL, flags|syscall.O_NONBLOCK)
    return err
}
```

### 8.3 批量处理

```go
// 批量唤醒 goroutine
func injectglist(glist *gList) {
    if glist.empty() {
        return
    }
    
    lock(&sched.lock)
    for !glist.empty() {
        gp := glist.pop()
        casgstatus(gp, _Gwaiting, _Grunnable)
        globrunqput(gp)
    }
    unlock(&sched.lock)
}
```

### 9.2 监控网络连接

```go
import "net"

func monitorConnections() {
    ln, _ := net.Listen("tcp", ":8080")
    
    for {
        conn, _ := ln.Accept()
        go handleConn(conn)
    }
}

func handleConn(conn net.Conn) {
    defer conn.Close()
    
    // 设置截止时间
    conn.SetDeadline(time.Now().Add(30 * time.Second))
    
    buf := make([]byte, 1024)
    n, err := conn.Read(buf)
    if err != nil {
        if netErr, ok := err.(net.Error); ok && netErr.Timeout() {
            fmt.Println("read timeout")
        }
        return
    }
    
    conn.Write(buf[:n])
}
```
