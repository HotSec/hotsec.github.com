# 类型检查

> 参考：《Go语言高级编程（第2版）》第5章

类型检查是 Go 编译器的重要阶段，负责验证程序的类型安全性。Go 是静态类型语言，编译器需要在编译期确定所有表达式的类型。

## 二、类型检查过程

### 2.1 类型检查入口

```go
func typecheck(n *Node, top int) *Node {
    if n == nil || n.Typecheck != 0 {
        return n
    }
    
    n.Typecheck = 1
    
    switch n.Op {
    case OLITERAL:
        n.Type = n.Val.Type()
        
    case ONAME:
        n.Type = n.Sym.Type
        
    case OTYPE:
        n.Type = types.New(n.Sym)
        
    case OPAREN:
        n.Left = typecheck(n.Left, top)
        n.Type = n.Left.Type
        
    case OADD, OSUB, OMUL, ODIV:
        n.Left = typecheck(n.Left, top)
        n.Right = typecheck(n.Right, top)
        n.Type = binaryType(n.Op, n.Left.Type, n.Right.Type)
        
    case OCALL:
        n = typecheckCall(n)
        
    case OAS:
        n = typecheckAssign(n)
    }
    
    return n
}
```

### 2.2 表达式类型检查

```go
func binaryType(op Op, l, r *Type) *Type {
    if l == nil || r == nil {
        return nil
    }
    
    switch op {
    case OADD, OSUB, OMUL, ODIV:
        if !types.IsNumeric[l.Etype] || !types.IsNumeric[r.Etype] {
            yyerror("invalid operation: %v (mismatched types)", op)
            return nil
        }
        return l
        
    case OEQ, ONE:
        if !types.Identical(l, r) {
            yyerror("invalid operation: %v (mismatched types)", op)
            return nil
        }
        return types.Types[TBOOL]
        
    case OLT, OLE, OGT, OGE:
        if !types.IsOrdered[l.Etype] || !types.IsOrdered[r.Etype] {
            yyerror("invalid operation: %v (mismatched types)", op)
            return nil
        }
        return types.Types[TBOOL]
    }
    
    return nil
}
```

### 2.3 函数调用类型检查

```go
func typecheckCall(n *Node) *Node {
    n.Left = typecheck(n.Left, ctxExpr|ctxType)
    
    var t *Type
    switch n.Left.Op {
    case OTYPE:
        return typecheckConversion(n)
    case OCALLFUNC, OCALLMETH:
        t = n.Left.Type
    }
    
    if t == nil {
        yyerror("cannot call non-function")
        return n
    }
    
    if !t.IsFunc() {
        yyerror("cannot call non-function %v", t)
        return n
    }
    
    params := t.Params()
    args := n.List
    
    if len(args) != len(params) {
        yyerror("not enough arguments in call to %v", n.Left)
        return n
    }
    
    for i, arg := range args {
        args[i] = typecheck(arg, ctxExpr)
        if !types.AssignableTo(args[i].Type, params[i].Type) {
            yyerror("cannot use %v as type %v", args[i].Type, params[i].Type)
        }
    }
    
    return n
}
```

## 三、类型推断

### 3.1 常量类型推断

```go
func constType(v Val) *Type {
    switch v.U.(type) {
    case bool:
        return types.Types[TBOOL]
    case string:
        return types.Types[TSTRING]
    case int64:
        return types.Types[TINT]
    case *big.Float:
        return types.Types[TFLOAT]
    case *big.Rat:
        return types.Types[TCOMPLEX]
    }
    return nil
}

func defaultType(t *Type) *Type {
    switch t {
    case types.UntypedBool:
        return types.Types[TBOOL]
    case types.UntypedString:
        return types.Types[TSTRING]
    case types.UntypedInt:
        return types.Types[TINT]
    case types.UntypedFloat:
        return types.Types[TFLOAT64]
    case types.UntypedComplex:
        return types.Types[TCOMPLEX128]
    case types.UntypedRune:
        return types.Types[TRUNE]
    }
    return t
}
```

### 3.2 赋值类型推断

```go
func assignType(dst, src *Type) *Type {
    if src == types.UntypedNil {
        if dst.HasNil() {
            return dst
        }
        yyerror("cannot use nil as type %v", dst)
        return nil
    }
    
    if src == types.UntypedBool {
        if dst.IsBoolean() {
            return dst
        }
        yyerror("cannot use bool as type %v", dst)
        return nil
    }
    
    if src == types.UntypedInt {
        if dst.IsInteger() {
            return dst
        }
        yyerror("cannot use int as type %v", dst)
        return nil
    }
    
    return src
}
```

### 3.3 泛型类型推断

Go 1.18+ 引入泛型后，类型检查需要处理类型参数：

```go
func instantiate(f *Func, targs []*Type) *Func {
    subst := make(map[*TypeParam]*Type)
    
    tparams := f.Type.Params().TypeParams()
    for i, tparam := range tparams {
        subst[tparam] = targs[i]
    }
    
    return substFunc(f, subst)
}

func substFunc(f *Func, subst map[*TypeParam]*Type) *Func {
    newf := f.copy()
    newf.Type = substType(f.Type, subst)
    return newf
}

func substType(t *Type, subst map[*TypeParam]*Type) *Type {
    if tparam, ok := t.(*TypeParam); ok {
        if subst, ok := subst[tparam]; ok {
            return subst
        }
    }
    
    switch t.Kind() {
    case TARRAY:
        return types.NewArray(substType(t.Elem(), subst), t.Len())
    case TSLICE:
        return types.NewSlice(substType(t.Elem(), subst))
    case TMAP:
        return types.NewMap(
            substType(t.Key(), subst),
            substType(t.Elem(), subst),
        )
    case TSTRUCT:
        return substStruct(t, subst)
    case TFUNC:
        return substFuncType(t, subst)
    }
    
    return t
}
```

## 四、类型兼容性

### 4.1 类型一致性

```go
func Identical(t1, t2 *Type) bool {
    if t1 == t2 {
        return true
    }
    if t1 == nil || t2 == nil {
        return false
    }
    
    if t1.Kind() != t2.Kind() {
        return false
    }
    
    switch t1.Kind() {
    case TARRAY:
        return t1.Len() == t2.Len() && Identical(t1.Elem(), t2.Elem())
    case TSLICE, TPTR:
        return Identical(t1.Elem(), t2.Elem())
    case TMAP:
        return Identical(t1.Key(), t2.Key()) && Identical(t1.Elem(), t2.Elem())
    case TSTRUCT:
        return identicalStructs(t1, t2)
    case TFUNC:
        return identicalFuncs(t1, t2)
    case TINTERFACE:
        return identicalInterfaces(t1, t2)
    }
    
    return true
}
```

### 4.2 可赋值性

```go
func AssignableTo(src, dst *Type) bool {
    if src == dst {
        return true
    }
    
    if src == types.UntypedNil {
        return dst.HasNil()
    }
    
    if src == types.UntypedBool {
        return dst.IsBoolean()
    }
    
    if src.IsUntyped() {
        return implicitlyConvertible(src, dst)
    }
    
    if implements(src, dst) {
        return true
    }
    
    if isCallable(src) && isCallable(dst) {
        return callableAssignable(src, dst)
    }
    
    return Identical(src, dst)
}
```

### 4.3 接口实现检查

```go
func implements(t, iface *Type) bool {
    if !iface.IsInterface() {
        return false
    }
    
    for _, m := range iface.AllMethods() {
        if !hasMethod(t, m) {
            return false
        }
    }
    return true
}

func hasMethod(t *Type, m *Field) bool {
    for _, tm := range t.AllMethods() {
        if tm.Name == m.Name && Identical(tm.Type, m.Type) {
            return true
        }
    }
    return false
}
```

## 五、类型转换

### 5.1 显式类型转换

```go
func convert(n *Node, t *Type) *Node {
    if n.Type == nil {
        return n
    }
    
    if !convertibleTo(n.Type, t) {
        yyerror("cannot convert %v to type %v", n.Type, t)
        return n
    }
    
    n = nod(OCALL, typename(t), n)
    n.Type = t
    return n
}

func convertibleTo(src, dst *Type) bool {
    if Identical(src, dst) {
        return true
    }
    
    if src.IsNumeric() && dst.IsNumeric() {
        return true
    }
    
    if src.IsString() && dst.IsSlice() && dst.Elem().Kind() == types.TUINT8 {
        return true
    }
    
    if src.IsSlice() && src.Elem().Kind() == types.TUINT8 && dst.IsString() {
        return true
    }
    
    return false
}
```

### 5.2 隐式类型转换

```go
func implicitConversion(src, dst *Type) bool {
    if src == types.UntypedNil {
        return dst.HasNil()
    }
    
    if src == types.UntypedBool {
        return dst.IsBoolean()
    }
    
    if src == types.UntypedString {
        return dst.IsString()
    }
    
    if src == types.UntypedInt {
        return dst.IsInteger() && represents(dst, src)
    }
    
    if src == types.UntypedFloat {
        return dst.IsFloat() && represents(dst, src)
    }
    
    if src == types.UntypedComplex {
        return dst.IsComplex() && represents(dst, src)
    }
    
    return false
}
```

## 六、类型断言

### 6.1 类型断言检查

```go
func typeAssert(n *Node, t *Type) *Node {
    if !n.Type.IsInterface() {
        yyerror("invalid type assertion: %v (non-interface type)", n.Type)
        return n
    }
    
    if t.IsInterface() {
        if !implements(n.Type, t) {
            yyerror("impossible type assertion: %v does not implement %v", n.Type, t)
        }
    } else {
        if !isConcreteType(t) {
            yyerror("invalid type assertion: %v (non-concrete type)", t)
        }
        if !implements(t, n.Type) {
            yyerror("impossible type assertion: %v does not implement %v", t, n.Type)
        }
    }
    
    return n
}
```

### 6.2 类型 switch

```go
func typeSwitch(n *Node) *Node {
    if !n.Tag.Type.IsInterface() {
        yyerror("cannot type switch on non-interface value")
        return n
    }
    
    for _, c := range n.List {
        if c.Op != OCASE {
            continue
        }
        
        for _, t := range c.List {
            if t.Op == OTYPE {
                checkTypeAssert(n.Tag, t.Type)
            } else if t.Op == ONIL {
                // nil case
            }
        }
    }
    
    return n
}
```

## 七、泛型类型检查

### 7.1 类型参数约束

```go
type TypeParam struct {
    Index int
    Bound *Type
}

func checkTypeParam(tparam *TypeParam, targ *Type) bool {
    if tparam.Bound == nil {
        return true
    }
    
    if tparam.Bound.IsInterface() {
        return implements(targ, tparam.Bound)
    }
    
    return Identical(targ, tparam.Bound)
}
```

### 7.2 类型实例化

```go
func instantiateType(t *Type, targs []*Type) *Type {
    if len(targs) == 0 {
        return t
    }
    
    subst := make(map[*TypeParam]*Type)
    tparams := t.TypeParams()
    
    for i, tparam := range tparams {
        if !checkTypeParam(tparam, targs[i]) {
            yyerror("%v does not implement %v", targs[i], tparam.Bound)
            return t
        }
        subst[tparam] = targs[i]
    }
    
    return substType(t, subst)
}
```

## 八、总结

类型检查是 Go 编译器的关键阶段，确保程序的类型安全性：

1. **静态类型检查**：编译期确定所有类型
2. **类型推断**：自动推断常量和表达式类型
3. **接口检查**：验证接口实现
4. **泛型支持**：Go 1.18+ 的类型参数和约束检查
5. **类型转换**：显式和隐式转换规则
