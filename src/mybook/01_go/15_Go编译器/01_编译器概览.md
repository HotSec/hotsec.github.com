# Go 编译器概览

> 参考：《Go语言高级编程（第2版）》第5章

Go 编译器是将 Go 源代码转换为机器码的核心工具。理解编译器的工作原理有助于编写更高效的代码，也能帮助开发者进行性能调优和问题排查。

---

### 1.2 编译器入口

Go 编译器的入口在 `cmd/compile` 目录：

```
cmd/compile/
├── main.go           # 编译器入口
├── internal/
│   ├── syntax        # 词法和语法分析
│   ├── gc            # 类型检查和语义分析
│   ├── ssa           # SSA 中间表示
│   ├── obj           # 目标文件生成
│   └── arm64/...     # 各架构后端
└── ...
```

### 1.3 编译阶段详解

| 阶段 | 输入 | 输出 | 主要工作 |
|------|------|------|---------|
| 词法分析 | 源代码 | Token 流 | 识别关键字、标识符、字面量 |
| 语法分析 | Token 流 | AST | 构建抽象语法树 |
| 类型检查 | AST | 类型信息 | 类型推断、类型一致性检查 |
| 语义分析 | AST + 类型 | IR | 作用域、变量捕获、逃逸分析 |
| SSA 生成 | IR | SSA IR | 生成静态单赋值形式 |
| 优化 | SSA IR | 优化后 SSA | 死代码消除、内联、常量传播 |
| 代码生成 | SSA IR | 汇编代码 | 指令选择、寄存器分配 |
| 链接 | 目标文件 | 可执行文件 | 符号解析、重定位 |

### 2.2 词法分析器实现

```go
type Scanner struct {
    source  []byte
    offset  int
    line    int
    column  int
}

func (s *Scanner) Scan() (token.Token, string) {
    s.skipWhitespace()
    
    if s.offset >= len(s.source) {
        return token.EOF, ""
    }
    
    ch := s.source[s.offset]
    
    switch {
    case isLetter(ch):
        return s.scanIdentifier()
    case isDigit(ch):
        return s.scanNumber()
    case ch == '"':
        return s.scanString()
    default:
        return s.scanOperator()
    }
}

func (s *Scanner) scanIdentifier() (token.Token, string) {
    start := s.offset
    for s.offset < len(s.source) && isLetter(s.source[s.offset]) {
        s.offset++
    }
    lit := string(s.source[start:s.offset])
    
    if tok := token.Lookup(lit); tok.IsKeyword() {
        return tok, lit
    }
    return token.IDENT, lit
}
```

## 三、语法分析

### 3.1 AST 节点类型

Go 的 AST 定义在 `go/ast` 包：

```go
type Node interface {
    Pos() token.Pos
    End() token.Pos
}

type Expr interface {
    Node
    exprNode()
}

type Stmt interface {
    Node
    stmtNode()
}

type Decl interface {
    Node
    declNode()
}

type File struct {
    Name  *Ident
    Decls []Decl
}

type FuncDecl struct {
    Recv *FieldList
    Name *Ident
    Type *FuncType
    Body *BlockStmt
}

type FuncType struct {
    Params  *FieldList
    Results *FieldList
}

type BlockStmt struct {
    List []Stmt
}

type IfStmt struct {
    Init Stmt
    Cond Expr
    Body *BlockStmt
    Else Stmt
}

type ForStmt struct {
    Init Stmt
    Cond Expr
    Post Stmt
    Body *BlockStmt
}

type RangeStmt struct {
    Key, Value Expr
    Tok        token.Token
    X          Expr
    Body       *BlockStmt
}

type CallExpr struct {
    Fun  Expr
    Args []Expr
}

type BinaryExpr struct {
    X     Expr
    Op    token.Token
    Y     Expr
}

type Ident struct {
    Name string
    Obj  *Object
}
```

### 3.2 语法分析器

```go
type Parser struct {
    scanner *Scanner
    tok     token.Token
    lit     string
}

func (p *Parser) ParseFile() *ast.File {
    p.expect(token.PACKAGE)
    name := p.parseIdent()
    
    var decls []ast.Decl
    for p.tok != token.EOF {
        decls = append(decls, p.parseDecl())
    }
    
    return &ast.File{
        Name:  name,
        Decls: decls,
    }
}

func (p *Parser) parseDecl() ast.Decl {
    switch p.tok {
    case token.IMPORT:
        return p.parseImportDecl()
    case token.CONST:
        return p.parseConstDecl()
    case token.TYPE:
        return p.parseTypeDecl()
    case token.VAR:
        return p.parseVarDecl()
    case token.FUNC:
        return p.parseFuncDecl()
    default:
        p.errorExpected(p.tok, "declaration")
        return nil
    }
}

func (p *Parser) parseFuncDecl() *ast.FuncDecl {
    p.expect(token.FUNC)
    
    recv := p.parseReceiver()
    name := p.parseIdent()
    typ := p.parseFuncType()
    body := p.parseBody()
    
    return &ast.FuncDecl{
        Recv: recv,
        Name: name,
        Type: typ,
        Body: body,
    }
}

func (p *Parser) parseExpr() ast.Expr {
    return p.parseBinaryExpr(token.LowestPrec + 1)
}

func (p *Parser) parseBinaryExpr(prec int) ast.Expr {
    x := p.parseUnaryExpr()
    
    for {
        op := p.tok
        opPrec := token.Precedence(op)
        if opPrec < prec {
            return x
        }
        
        p.next()
        y := p.parseBinaryExpr(opPrec + 1)
        x = &ast.BinaryExpr{
            X:  x,
            Op: op,
            Y:  y,
        }
    }
}
```

## 五、编译器优化

### 5.1 内联优化

```go
func add(a, b int) int {
    return a + b
}

func main() {
    result := add(1, 2)
    println(result)
}
```

查看内联决策：

```bash
go build -gcflags="-m=2" main.go
```

输出：
```
./main.go:3:6: can inline add with cost 4 as: func(int, int) int { return a + b }
./main.go:7:13: inlining call to add func(int, int) int { return a + b }
```

### 5.2 逃逸分析

```go
func escape() *int {
    x := 1
    return &x
}

func noEscape() int {
    x := 1
    return x
}
```

```bash
go build -gcflags="-m" main.go
```

输出：
```
./main.go:2:2: moved to heap: x
```

### 5.3 边界检查消除

```go
func sum(s []int) int {
    total := 0
    for i := 0; i < len(s); i++ {
        total += s[i]
    }
    return total
}

func sumBCE(s []int) int {
    total := 0
    _ = s[len(s)-1]
    for i := 0; i < len(s); i++ {
        total += s[i]
    }
    return total
}
```

```bash
go build -gcflags="-d=ssa/check_bce/debug=1" main.go
```

## 六、编译器扩展

### 6.1 自定义分析工具

```go
package main

import (
    "fmt"
    "go/ast"
    "go/parser"
    "go/token"
)

type FuncVisitor struct {
    fset *token.FileSet
}

func (v *FuncVisitor) Visit(node ast.Node) ast.Visitor {
    if fn, ok := node.(*ast.FuncDecl); ok {
        pos := v.fset.Position(fn.Pos())
        fmt.Printf("function %s at %s\n", fn.Name.Name, pos)
    }
    return v
}

func main() {
    fset := token.NewFileSet()
    f, err := parser.ParseFile(fset, "main.go", nil, 0)
    if err != nil {
        panic(err)
    }
    
    ast.Walk(&FuncVisitor{fset: fset}, f)
}
```

## 八、总结

Go 编译器是一个复杂的系统，理解其工作原理有助于：

1. **编写高效代码**：理解逃逸分析、内联决策
2. **性能调优**：使用 SSA 查看器分析优化机会
3. **问题排查**：通过查看汇编定位性能问题
4. **工具开发**：基于 AST 开发代码分析工具
