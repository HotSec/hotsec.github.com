      defer func() {
          if r := recover(); r != nil {
              switch x := r.(type) {
              case string:
                  err = errors.New(x)
              case error:
                  err = x
              default:
                  err = fmt.Errorf("Unknown panic: %v", r)
              }
          }
      }()
