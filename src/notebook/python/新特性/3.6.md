## 类型注解

```python
def greet(name: str) -> str:
    return 'Hello ' + name
```

## 异步生成器

```python
async def countdown(n):
    while n > 0:
        yield n
        n -= 1
        await asyncio.sleep(1)
```

## 异步上下文管理器

```python
async def get(file):
    async with aiofiles.open(file, mode='r') as f:
        return await f.read()

async def put(file, data):
    async with aiofiles.open(file, mode='w') as f:
        await f.write(data)
```

## 异步的 `await` 和 `async`

```python
async def main():
    await asyncio.sleep(1)
    print('Hello')
    await asyncio.sleep(1)
    print('World')

asyncio.run(main())
```

## `asyncio.run`

```python
import asyncio

async def main():
    print('Hello')
    await asyncio.sleep(1)
    print('World')

asyncio.run(main())
```

## `asyncio.create_task`

```python
import asyncio

async def main():
    task1 = asyncio.create_task(some_coro())

    task2 = asyncio.create_task(some_other_coro())

    await task1
    await task2

asyncio.run(main())
```

## `asyncio.gather`

```python
import asyncio

async def main():
    await asyncio.gather(some_coro(), some_other_coro())

asyncio.run(main())
```

## `asyncio.wait`

```python
import asyncio

async def main():
    done, pending = await asyncio.wait(
        [some_coro(), some_other_coro()],
        timeout=1.0,
    )

asyncio.run(main())
```
