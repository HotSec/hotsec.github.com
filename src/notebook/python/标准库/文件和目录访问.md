# 文件和目录访问

```
from pathlib import Path
p = Path('.')

# 列出子目录
[x for x in p.iterdir() if x.is_dir()]

# 列出当前目录下所有的.py文件
list(p.glob('**/*.py'))

# p = Path('/etc')
q = p / 'init.d' / 'reboot'
# PosixPath('/etc/init.d/reboot')
q.resolve()
# PosixPath('/etc/rc.d/init.d/halt')
q.exists() # True
q.is_dir() # False
```
